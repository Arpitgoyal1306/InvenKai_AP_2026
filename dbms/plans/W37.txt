UPDATE orders
       SET total_amount = (
         SELECT COALESCE(SUM(line_total), 0)
         FROM order_items
         WHERE order_id = $1
       ),
       updated_at = CURRENT_TIMESTAMP
       WHERE id = $1
       RETURNING id, user_id, status, total_amount,
                 created_at, updated_at;
