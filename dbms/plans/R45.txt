R45 · Buyer order page and item counts

BEGIN;

PREPARE sample AS
SELECT
       o.id,
       o.user_id,
       o.status,
       o.total_amount,
       o.created_at,
       o.updated_at,
       COUNT(oi.id)::INTEGER AS item_count
     FROM orders o
     LEFT JOIN order_items oi
       ON oi.order_id = o.id
     WHERE o.user_id = $1
     GROUP BY o.id
     ORDER BY o.created_at DESC, o.id DESC
     LIMIT $2 OFFSET $3;
EXPLAIN (ANALYZE, BUFFERS) EXECUTE sample(33,20,0);
